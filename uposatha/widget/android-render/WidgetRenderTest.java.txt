package gift.dhamma.uposatha;

import android.content.Context;
import android.graphics.Bitmap;

import org.json.JSONObject;
import org.junit.Test;
import org.junit.runner.RunWith;
import org.robolectric.RobolectricTestRunner;
import org.robolectric.RuntimeEnvironment;
import org.robolectric.annotation.Config;
import org.robolectric.annotation.GraphicsMode;

import java.io.File;
import java.io.FileOutputStream;
import java.nio.file.Files;

@RunWith(RobolectricTestRunner.class)
@GraphicsMode(GraphicsMode.Mode.NATIVE)
@Config(sdk = 34)
public class WidgetRenderTest {
    static final long H = 3600000L;

    @Test
    public void renderAll() throws Exception {
        Context ctx = RuntimeEnvironment.getApplication();
        String out = System.getProperty("widget.out", "/tmp/wd/out");
        new File(out).mkdirs();
        String base = new String(Files.readAllBytes(new File("/var/www/dg-apps/uposatha/widget/fixture-ru.json").toPath()), "UTF-8");
        // name, lang, detail, showKala, placeSet, south, nowKind(0 gen,1 sunrise+3h,2 upo[1]+5h)
        Object[][] sc = {
                { "a", "ru", false, true, false, false, 0 },
                { "b", "ru", true, true, true, false, 1 },
                { "c", "ru", true, true, true, false, 2 },
                { "d", "en", false, false, true, false, 1 },
                { "e", "en", true, true, true, true, 0 },
        };
        int[][] sizes = { { 160, 160 }, { 360, 130 }, { 360, 300 }, { 360, 382 } };
        String[] names = { "s", "m", "l", "xl" };
        for (Object[] c : sc) {
            JSONObject d = new JSONObject(base);
            d.getJSONObject("settings").put("lang", c[1]).put("detail", c[2]).put("showKala", c[3]).put("placeSet", c[4]).put("south", c[5]);
            long now = 1791558373393L;
            if ((int) c[6] == 1) now = d.getJSONArray("days").getJSONObject(0).getJSONObject("sunrise").getLong("ms") + 3 * H;
            if ((int) c[6] == 2) now = d.getJSONArray("uposathas").getJSONObject(1).getJSONObject("start").getLong("ms") + 5 * H;
            for (int z = 0; z < sizes.length; z++) for (int ly = 0; ly < 3; ly++) for (int dk = 0; dk < 2; dk++) {
                Bitmap b = WidgetRenderer.render(ctx, d, ly, sizes[z][0], sizes[z][1], dk == 1, now, 2f);
                try (FileOutputStream fo = new FileOutputStream(out + "/" + c[0] + "-" + names[z] + "-l" + ly + (dk == 1 ? "-dark" : "-light") + ".png")) { b.compress(Bitmap.CompressFormat.PNG, 100, fo); }
            }
        }
        for (int k = 0; k < 2; k++) for (int z = 0; z < sizes.length; z++) {
            JSONObject d = k == 0 ? null : new JSONObject(base);
            Bitmap b = WidgetRenderer.render(ctx, d, 0, sizes[z][0], sizes[z][1], k == 1, 1791558373393L + 30 * 24 * H, 2f);
            try (FileOutputStream fo = new FileOutputStream(out + "/ph" + k + "-" + names[z] + ".png")) { b.compress(Bitmap.CompressFormat.PNG, 100, fo); }
        }
    }
}
