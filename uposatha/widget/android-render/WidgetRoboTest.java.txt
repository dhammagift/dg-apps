package gift.dhamma.uposatha;

import android.app.Application;
import android.graphics.Bitmap;
import android.view.View;
import android.widget.RemoteViews;

import org.junit.Test;
import org.junit.runner.RunWith;
import org.robolectric.RobolectricTestRunner;
import org.robolectric.RuntimeEnvironment;
import org.robolectric.annotation.Config;
import org.robolectric.annotation.GraphicsMode;

import java.io.File;

@RunWith(RobolectricTestRunner.class)
@GraphicsMode(GraphicsMode.Mode.NATIVE)
@Config(sdk = 34, qualifiers = "w411dp-h891dp-xxhdpi")
public class WidgetRoboTest {
    @Test
    public void render() throws Exception {
        Application ctx = RuntimeEnvironment.getApplication();
        File out = new File(System.getProperty("widget.out", "/tmp/claude-0/s/out"));
        out.mkdirs();
        String only = String.valueOf(System.getenv("WIDGET_ONLY") == null ? "" : System.getenv("WIDGET_ONLY"));
        for (int si = 0; si < WidgetSheet.SCENARIOS.length; si++) {
            WidgetSheet.Scenario sc = WidgetSheet.SCENARIOS[si];
            for (int[] sz : WidgetSheet.SIZES) {
                for (int dark = 0; dark < 2; dark++) {
                    for (int ly = 0; ly < 3; ly++) {
                        if (sc.file == null && ly > 0) continue;
                        for (int variant = 0; variant < (sc.file == null ? 2 : 1); variant++) {
                            String n = WidgetSheet.name(si, ly, sz[0], sz[1], dark == 1, variant);
                            if (!only.isEmpty() && !n.matches(only)) continue;
                            try {
                                RemoteViews rv = WidgetSheet.views(ctx, sc, variant, ly, sz[0], sz[1]);
                                View v = WidgetSheet.inflate(ctx, rv, sz[0], sz[1], dark == 1);
                                Bitmap b = WidgetSheet.draw(v, "#" + si + " L" + ly + " " + sz[0] + "x" + sz[1] + " dp " + (dark == 1 ? "dark" : "light"));
                                WidgetSheet.save(b, new File(out, n + ".png"));
                            } catch (Throwable t) {
                                System.out.println("FAILED " + n + ": " + t);
                                t.printStackTrace(System.out);
                            }
                        }
                    }
                }
            }
        }
    }
}
